function engine() {
  return new InterestEngineService(
    new PolicyGateV2Service(), new SegmentBuilderService(),
    new AllocationEngineService(new TBK100AllocatorService(), new ClaimPriorityService()),
    {} as never, { record: () => undefined, clearAll: () => undefined } as never, new VersionPinningService(), undefined,
  );
}
const d = (s: string | null | undefined) => (s ? new Date(`${s}T00:00:00.000Z`) : null);
const item = (p: Record<string, unknown>) => ({
  itemType: 'PRINCIPAL', currency: 'TRY', status: 'ACTIVE', metadata: null, interestType: null, interestTypeCode: null,
  interestRate: null, interestStartDate: null, interestAccrualStatus: null, interestStartDateProvenance: null, ...p,
  demandedAmount: p.amount, interestStartDate: d(p.start as string),
});
const col = (id: string, amount: number, date: string) => ({ id, status: 'CONFIRMED', cancelledAt: null, amount, currency: 'TRY', date: d(date), sourceType: 'BANKA', channel: null });
const legal = (from: string, to: string | null, rate: number, id = from) => ({

  source: RateSourceType.TCMB, versionHash: 'h', createdAt: '2025-01-01T00:00:00Z',
});

async function run(caseRow: any, items: any[], collections: any[], rates: any[], asOf: string, overpayments: any[] = []) {
  const prisma: any = {
    case: { findFirst: async () => caseRow },
    claimItem: { findMany: async () => items },
    ledgerEntry: { findMany: async () => [] },
    collection: { findMany: async () => collections },
    collectionOverpayment: { findMany: async () => overpayments },
    icrabotTimelineEntry: { findMany: async () => [] },
  };
  const svc = new CaseBalanceService(prisma, { getRatesForPeriod: async () => rates } as never, engine());
  const bal = await svc.computeCaseBalance('t', 'c', asOf);
  const disp = toCaseBalanceDisplay({ tenantId: 't', caseId: 'c', balance: bal, generatedAt: '2026-09-30T00:00:00.000Z' });
  const r = bal.currencyResults[0]?.result ?? null;
  return {
    status: disp.status, fatal: bal.diagnostics.fatal.map((f) => f.code), skipped: bal.currencyResults.map((c) => c.skippedReason ?? null),
    totalDue: r?.totalDue ?? null, totalInterest: r?.totalInterest ?? null,
    finalStates: (r?.finalDebtStates ?? []).map((s: any) => ({ id: s.claimId, principal: s.principal, accruedInterest: s.accruedInterest })),
    perBucketInterest: (r?.segments ?? []).reduce((acc: any, s: any) => { acc[s.claimBucketId] = Math.round(((acc[s.claimBucketId] ?? 0) + s.interestAmount) * 100) / 100; return acc; }, {}),
    policyWarnings: (r?.policyWarnings ?? []).map((w: any) => w.code ?? w),
    allocations: (r?.allocations ?? []).map((a: any) => ({ p: a.paymentId, to: a.claimBucketId, a: (a.allocations ?? []).map((x: any) => `${x.category ?? x.type}:${x.amountAllocated}`) })),
    unsimulated: bal.unsimulatedPrincipals ?? null,
    totals: disp.totals, costs: disp.costs, ancillaries: disp.ancillaries,
  };
}
